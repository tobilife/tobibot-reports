// Agent Runtime Orchestration (TypeScript + Bun + Claude MCP Engine)
import { AgentOrchestrator, ToolRegistry } from '@tobilife/agent-core';
import { createClient } from '@supabase/supabase-js';

export class AutonomousWorker {
  private registry = new ToolRegistry();
  private db = createClient(process.env.SUPABASE_URL!, process.env.SERVICE_ROLE_KEY!);

  async dispatchTask(prompt: string, context: Record<string, unknown>) {
    // High-throughput parallel inference with recursive reflection loop
    const plan = await this.registry.decompose(prompt, { maxDepth: 3 });
    const result = await AgentOrchestrator.executeStream(plan, {
      telemetry: true,
      onToken: (chunk) => process.stdout.write(chunk)
    });
    return result;
  }
}

  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  workspace_id uuid REFERENCES workspaces(id) ON DELETE CASCADE,
  embedding vector(1536) NOT NULL,
  metadata jsonb DEFAULT '{}'::jsonb,
  created_at timestamptz DEFAULT now()
);

-- Enforce Tenant Isolation via Strict RLS Policy
ALTER TABLE public.agent_memory_nodes ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Tenant Isolated Access"
  ON public.agent_memory_nodes
  FOR ALL
  USING (workspace_id = (current_setting('request.jwt.claims', true)::jsonb ->> 'workspace_id')::uuid);
