빠르고 직관적인 SDK 사용법

단 몇 줄의 코드로 당신의 자율 에이전트 서비스에 멀티 채널 브릿지를 장착하세요.

server.ts 
import { AgentBridge } from '@tobilife/agent-messenger-bridge';

// 1. 브릿지 인스턴스 초기화
const bridge = new AgentBridge({
  apiKey: process.env.BRIDGE_API_KEY,
  adapters: ['slack', 'telegram', 'discord'],
});

// 2. 정규화된 메시지 이벤트 리스닝
bridge.onMessage(async ({ channel, content, reply }) => {
  const response = await aiAgent.generate({ prompt: content });
  await reply.stream(response);
});

await bridge.start();
