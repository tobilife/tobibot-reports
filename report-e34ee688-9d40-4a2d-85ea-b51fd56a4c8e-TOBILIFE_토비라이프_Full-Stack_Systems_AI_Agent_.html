// 01 Architecture: Autonomous Multi-Agent State Machine
import { AgentRuntime, ToolRegistry } from '@tobilife/agent-core';
import { PostgresPool } from '@supabase/ssr';

export class SystemOrchestrator {
  private runtime: AgentRuntime;

  constructor(env: EnvConfig) {
    this.runtime = new AgentRuntime({
      model: 'gemini-2.5-pro',
      vectorStore: env.VECTOR_DB_URL,
      memoryPolicy: 'isolated-session-with-rollup',
    });
  }

  public async dispatchPipeline(task: JobPayload): Promise<ExecutionResult> {
    const plan = await this.runtime.planTask(task.objective);
    return await this.runtime.executeWithRecovery(plan, { maxRetries: 3 });
  }
}
